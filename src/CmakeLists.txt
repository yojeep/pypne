set(pybind11_DIR ${CMAKE_SOURCE_DIR}/thirdparty/system/pybind11/share/cmake/pybind11)
find_package(pybind11 REQUIRED)



add_library(pypne_core STATIC
    pnextract/blockNet.cpp
    pnextract/medialSurf.cpp
    pnextract/pypne.cpp
)

target_link_libraries(pypne_core 
    PUBLIC
    pybind11::module  
)
target_include_directories(pypne_core
    PUBLIC  ${CMAKE_SOURCE_DIR}/thirdparty/system
    PUBLIC  ${CMAKE_SOURCE_DIR}/thirdparty/system/eigen-5.0.1
    PUBLIC  ${CMAKE_SOURCE_DIR}/thirdparty/system/eigen-5.0.1/unsupported
    PUBLIC  ${CMAKE_SOURCE_DIR}/thirdparty/system/sdsl-lite-3.0.4/include
)

target_compile_definitions(pypne_core PRIVATE BOOST_DISABLE_ASSERTS)


if(MSVC)
    target_compile_options(pypne_core PRIVATE /O2 /W1 /GL /LTCG /MT)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(pypne_core PRIVATE -O3 -Wall)
endif()
